5.4 Escalation Ladder
// Sys/Kernel/Gardener/Escalation.halo
fn next(inc: Incident) raises {
    switch (inc.escalation) {
        case 0: HotPatch.rollback(inc.symbol) raises;                       // last known good fn
        case 1: Modules.reload(inc.symbol.module, from = Grove.head("/Sys")) raises;
        case 2: Grove.revert("/Sys", to = Memory.last_stable_snapshot()); Modules.reload_all();
        case 3: Pstore.write(inc); Arch.reboot_into(Slot.current, "--resume-incident");
        case 4: Slot.mark_crash(); Arch.reboot_into(Slot.other);           // watchdog would do this anyway
        default: throw 'Exhausted';                                        // logged; agent keeps running
    }
    inc.escalation += 1;
    Gardener.incidents.push(inc);       // the *result* of escalating is re-examined by a planner
}
On --resume-incident, the first thing AgentLoop.run does is load Pstore and push those incidents at top priority. Correction resumes exactly where it left off, in the new kernel instance.
